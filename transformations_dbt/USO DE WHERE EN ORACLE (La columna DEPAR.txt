# CANTIDAD DE EMPRESAS POR DEPARTAMENTO CON SU PROMEDIO DE INGRESOS
SELECT DEPARTAMENTO_DOMICILIO,COUNT(DEPARTAMENTO_DOMICILIO) AS EMPRESAS_XDEPARTAMENTO,AVG(INGRESOS_OPERACIONALES) AS PROMEDIO_INGRESOS
FROM TOP10000
WHERE MACROSECTOR = 'MANUFACTURA'
GROUP BY DEPARTAMENTO_DOMICILIO
HAVING AVG(INGRESOS_OPERACIONALES)> 0.1
ORDER BY PROMEDIO_INGRESOS DESC
FETCH FIRST 10 ROWS ONLY;

      FROM TOP10000
      WHERE MACROSECTOR = 'CONSTRUCCIÓN'
      GROUP BY DEPARTAMENTO_DOMICILIO)

#INGRESOS MAYORES AL PROMEDIO POR DEPARTAMENTO DE SECTOR COMERCIO (CROSS JOIN PARA UNIR TABLAS SIN COLUMNAS RELACIONADAS)

WITH DEPARTAMENTOS AS (SELECT DEPARTAMENTO_DOMICILIO, SUM(INGRESOS_OPERACIONALES) AS TOTAL_INGRESOS_OPERACIONALES
FROM TOP10000
WHERE MACROSECTOR = 'COMERCIO'
GROUP BY DEPARTAMENTO_DOMICILIO),

PROMEDIO AS (SELECT AVG(TOTAL_INGRESOS_OPERACIONALES) AS PROMEDIO_TOTAL
FROM DEPARTAMENTOS)

SELECT D.DEPARTAMENTO_DOMICILIO, D.TOTAL_INGRESOS_OPERACIONALES
FROM DEPARTAMENTOS D
CROSS JOIN PROMEDIO P
WHERE D.TOTAL_INGRESOS_OPERACIONALES > P.PROMEDIO_TOTAL
ORDER BY TOTAL_INGRESOS_OPERACIONALES DESC

SUM(INGRESOS_OPERACIONALES) 
OVER(PARTITION BY DEPARTAMENTO_DOMICILIO ORDER BY INGRESOS_OPERACIONALES DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS ACUMULACION_DEPARTAMENTO FROM TOP10000

## DIVISIÓN EN 4 PARTES 'IGUALES' DE CADA MACROSECTOR(NTILE)

SELECT RAZÓN_SOCIAL, MACROSECTOR, INGRESOS_OPERACIONALES, 
NTILE(4) OVER(PARTITION BY MACROSECTOR ORDER BY INGRESOS_OPERACIONALES DESC) AS TOPS_XMACROSECTOR FROM TOP10000

## TOP10 EMPRESAS COLOMBIANAS POR INGRESOS (DENSE_RANK)

WITH EMPRESAS AS (SELECT RAZÓN_SOCIAL, INGRESOS_OPERACIONALES, DENSE_RANK() OVER(ORDER BY INGRESOS_OPERACIONALES DESC) AS EMPRESAS_TOP
FROM TOP10000)

SELECT * FROM EMPRESAS
WHERE EMPRESAS_TOP <= 10

## MACROSECTORES POR ENCIMA DEL 80% DEL PROMEDIO GLOBAL

WITH PROMEDIO AS (SELECT MACROSECTOR,
AVG(INGRESOS_OPERACIONALES) OVER(PARTITION BY MACROSECTOR) PROMEDIO_MACROSECTOR,
AVG(INGRESOS_OPERACIONALES) OVER() AS PROMEDIO_GLOBAL
FROM TOP10000)

SELECT DISTINCT(MACROSECTOR),PROMEDIO_MACROSECTOR,PROMEDIO_GLOBAL FROM PROMEDIO
WHERE (PROMEDIO_MACROSECTOR/PROMEDIO_GLOBAL)> 0.8

## PARETO 80/20 (PORCENTAJE ACUMULADO DE INGRESOS POR MACROSECTOR AL 80%) 

WITH TOPN AS (SELECT RAZÓN_SOCIAL, MACROSECTOR, INGRESOS_OPERACIONALES, SUM(INGRESOS_OPERACIONALES) 
OVER(PARTITION BY MACROSECTOR ORDER BY INGRESOS_OPERACIONALES DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS INGRESOS_ACU,
SUM(INGRESOS_OPERACIONALES) OVER(PARTITION BY MACROSECTOR) AS TOTAL_XMACROSECTOR, DENSE_RANK() 
OVER(PARTITION BY MACROSECTOR ORDER BY INGRESOS_OPERACIONALES DESC) AS RANKING
FROM TOP10000),

TOP80 AS (SELECT RAZÓN_SOCIAL, MACROSECTOR, INGRESOS_OPERACIONALES, INGRESOS_ACU, TOTAL_XMACROSECTOR,
(INGRESOS_ACU/TOTAL_XMACROSECTOR)*100 AS PORCENTAJE_ACUMULADO, RANKING
FROM TOPN
ORDER BY MACROSECTOR, RANKING)

SELECT * FROM TOP80
WHERE PORCENTAJE_ACUMULADO <= 80
